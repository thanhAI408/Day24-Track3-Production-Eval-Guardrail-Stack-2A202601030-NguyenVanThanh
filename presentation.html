📐 Công thức Cohen's κ

def cohen_kappa(judge_labels, human_labels): n = len(judge_labels) p_o = sum(j == h for j, h in zip(judge_labels, human_labels)) / n # observed agreement p_e = (judge_labels.count(1)/n * human_labels.count(1)/n + # expected by chance judge_labels.count(0)/n * human_labels.count(0)/n) κ = (p_o - p_e) / (1 - p_e) # Cohen's kappa return κ
